       SUM(DATA_LENGTH) AS data_bytes, SUM(INDEX_LENGTH) AS index_bytes,
       SUM(DATA_LENGTH + INDEX_LENGTH) AS data_plus_indexes_bytes,
       ROUND(100.0 * SUM(INDEX_LENGTH) / NULLIF(SUM(DATA_LENGTH), 0), 2)
         AS index_to_data_percent
