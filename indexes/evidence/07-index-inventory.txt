       GROUP_CONCAT(COLUMN_NAME ORDER BY SEQ_IN_INDEX) AS columns_in_order

GROUP BY TABLE_NAME, INDEX_NAME, INDEX_TYPE ORDER BY TABLE_NAME, INDEX_NAME
